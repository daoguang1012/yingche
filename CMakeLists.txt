cmake_minimum_required(VERSION 3.21)
project(BeautyPlayer VERSION 0.6.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL Network)
set(GPUPIXEL_BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GPUPIXEL_ENABLE_FACE_DETECTOR OFF CACHE BOOL "" FORCE)
set(GPUPIXEL_BUILD_DESKTOP_DEMO OFF CACHE BOOL "" FORCE)
set(GPUPIXEL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/gpupixel EXCLUDE_FROM_ALL)
get_target_property(gpu_sources gpupixel SOURCES)
list(FILTER gpu_sources EXCLUDE REGEX "/core/gpupixel_context\\.cc$")
set_property(TARGET gpupixel PROPERTY SOURCES ${gpu_sources} "${CMAKE_SOURCE_DIR}/src/gpupixel_host_context.cpp")
set_target_properties(gpupixel PROPERTIES AUTOMOC OFF AUTORCC OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/app" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/app")
target_compile_definitions(gpupixel PRIVATE STBI_WINDOWS_UTF8 NOMINMAX)

qt_add_executable(BeautyPlayer WIN32 src/main.cpp src/player.cpp src/player.h src/video.cpp src/video.h src/mpv_api.h src/settings.h src/gpupixel_pipeline.cpp src/gpupixel_pipeline.h resources.qrc)
target_include_directories(BeautyPlayer PRIVATE tools/mpv/include third_party/gpupixel/third_party/glad/include third_party/gpupixel/third_party/glfw/include)
target_link_libraries(BeautyPlayer PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL gpupixel::gpupixel glad::glad opengl32)
target_compile_options(BeautyPlayer PRIVATE /utf-8 /W4)
set_target_properties(BeautyPlayer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/app" OUTPUT_NAME "BeautyPlayerV060")

add_executable(Yingche WIN32 src/launcher.cpp)
set_target_properties(Yingche PROPERTIES AUTOMOC OFF AUTORCC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/work/v060" OUTPUT_NAME "Yingche")
target_compile_options(Yingche PRIVATE /utf-8 /W4)

# Player-free background service for the browser extension: tray app plus its root launcher.
qt_add_executable(YingcheBridge WIN32 src/bridge_service.cpp)
target_link_libraries(YingcheBridge PRIVATE Qt6::Widgets Qt6::Network)
target_compile_options(YingcheBridge PRIVATE /utf-8 /W4)
set_target_properties(YingcheBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/app")

add_executable(YingcheService WIN32 src/launcher.cpp)
target_compile_definitions(YingcheService PRIVATE YINGCHE_SERVICE_LAUNCHER)
set_target_properties(YingcheService PROPERTIES AUTOMOC OFF AUTORCC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/work/v060" OUTPUT_NAME "YingcheService")
target_compile_options(YingcheService PRIVATE /utf-8 /W4)
